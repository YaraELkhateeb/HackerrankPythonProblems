def minion_game(string):
    vowels=['I','A','E','U','O']
    Kevin=0
    Stuart =0

    for i in range(len(string)) :
        if(string[i]in vowels):
            Kevin+=len(string)-i
        else:
            Stuart+=len(string)-i
        
                
    print(f"Kevin {Kevin}") if Kevin>Stuart else print(f"Stuart {Stuart}") if Stuart>Kevin else print("Draw")
            
            
    
if __name__ == '__main__':
    s = input()
    minion_game(s)
