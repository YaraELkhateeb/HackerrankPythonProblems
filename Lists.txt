if __name__ == '__main__':
    N = int(input())
list1=list()
i=0

while(i<N):
    query=input().split(" ")
    if(query[0]=="insert"):
        list1.insert(int(query[1]),int(query[2]))
    elif(query[0]=="print"):
        print(list1)
    elif(query[0]=="remove"):
        list1.remove(int(query[1]))
    elif(query[0]=="append"):
        list1.append(int(query[1]))
    elif(query[0]=="pop"):
        list1.pop( )
    elif(query[0]=="reverse"):
        list1.reverse()
    elif(query[0]=="sort"):
        list1.sort()
    i+=1
