def solve(s):
    l=list(s.replace(" ",",#,").replace(",,",",").split(","))

    sum=0

        sum+=1

            temp=list(i)
            temp[0]=temp[0].upper()
            k=k+("".join(temp))
        elif i[0].isalnum():
            k=k+("".join(i))
            
        else:
            k=k+(" ")  
             
    return(((k)))
if __name__ == '__main__':
