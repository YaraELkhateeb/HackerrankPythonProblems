import textwrap

def wrap(string, max_width):
    temp=str()
    words=str()
    for i in range(0,len(string),max_width):
        temp=string[i:i+max_width]
        print(temp)
            
    words.join(temp)
    return words

if __name__ == '__main__':
    string, max_width = input(), int(input())
    result = wrap(string, max_width)
    print(result)
